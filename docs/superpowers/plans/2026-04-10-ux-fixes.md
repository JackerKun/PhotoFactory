### Task 1: Add disk cache for face thumbnails in Rust

**Files:**
- Modify: `src-tauri/src/commands/faces.rs:168-186`
- Modify: `src-tauri/src/commands/project.rs` (AppState — check if `faces_cache_path` exists or add face_thumbnails dir)
- Modify: `src/api/tauri.ts:138-140`

The current `cmd_get_cluster_face_thumbnail` generates JPEG bytes and returns a base64 data URL every time. We'll write the JPEG to `.pf/ai_cache/face_thumbnails/` on first call and return the file path on subsequent calls, so the frontend can use `convertFileSrc` like regular thumbnails.

- [ ] **Step 1: Modify `cmd_get_cluster_face_thumbnail` to cache on disk**

Replace the current implementation in `src-tauri/src/commands/faces.rs`:

```rust
/// Get a face thumbnail for a cluster, caching the JPEG to disk.
#[tauri::command]
pub async fn cmd_get_cluster_face_thumbnail(cluster_id: i64, state: State<'_, AppState>) -> AppResult<String> {
    let conn = get_conn(state.clone())?;
    let project = {
        let guard = state.project.lock().unwrap();
        guard.as_ref().ok_or_else(|| AppError::Project("No project open".into()))?.clone()
    };

    // Get the first face assigned to this cluster
    let faces = face_dao::get_faces_for_person(&conn, cluster_id, 1)?;
    let face = faces.into_iter().next().ok_or_else(|| AppError::NotFound("No faces in this cluster".into()))?;

    // Get the photo path
    let photo = photo_dao::get_photo_by_id(&conn, face.photo_id)?
        .ok_or_else(|| AppError::NotFound(format!("Photo {} not found", face.photo_id)))?;

    // Build cache path: .pf/ai_cache/face_thumbnails/
    let cache_dir = project.ai_cache_path().join("face_thumbnails");
    std::fs::create_dir_all(&cache_dir)?;
    let thumb_path = cache_dir.join(format!("cluster_{}.jpg", cluster_id));

    // Return cached file if it exists
    if thumb_path.exists() {
        return Ok(thumb_path.to_string_lossy().to_string());
    }

    // Generate and save
    let photo_path = std::path::Path::new(&photo.path);
    let jpeg_bytes = generator::generate_face_thumbnail(
        photo_path,
        face.bbox_x as u32, face.bbox_y as u32,
        face.bbox_w as u32, face.bbox_h as u32,
        160,
    )?;
    std::fs::write(&thumb_path, &jpeg_bytes)?;

    Ok(thumb_path.to_string_lossy().to_string())
}
```

### Task 3: Add thumbnail lazy loading via IntersectionObserver in PhotoCard

**Files:**
- Modify: `src/components/PhotoCard.tsx`

The current PhotoCard fires thumbnail generation for ALL visible cards immediately. With 500 photos in "original" mode, this causes 500 synchronous thumbnail generations = CPU spike.

- [ ] **Step 1: Add IntersectionObserver to PhotoCard**

Replace the thumbnail loading logic in `PhotoCard.tsx`:

```tsx
import React, { useEffect, useState, useCallback, useRef } from "react";

export default React.memo(function PhotoCard({ photo, onClick, onDoubleClick, size, height, shape = "square", isSelected = false }: Props) {
  const [thumbUrl, setThumbUrl] = useState<string | null>(null);
  const [menuPos, setMenuPos] = useState<{ x: number; y: number } | null>(null);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;

    const observer = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          if (entry.isIntersecting && !thumbUrl) {
            observer.unobserve(node);
            let cancelled = false;
            api.getThumbnailPath(photo.id!, "small").then((path) => {
              if (!cancelled) setThumbUrl(convertFileSrc(path));
            }).catch(() => {
              if (!cancelled && photo.file_size <= 30 * 1024 * 1024) {
                api.getPhotoDataUrl(photo.path).then(setThumbUrl).catch(() => {});
              }
            });
            return;
          }
        }
      },
      { rootMargin: "200px" },
    );

    observer.observe(node);
    return () => { observer.disconnect(); };
  }, [photo.id, photo.path, photo.file_size, thumbUrl]);

  // ... rest of component unchanged, but add ref to root div:
  return (
    <div
      ref={ref}
      className={`...`}
      // ... rest unchanged
```

This defers thumbnail loading until the card is within 200px of the viewport. Cards off-screen won't trigger any I/O or image processing.

pub fn get_label_preview_photo_id(conn: &rusqlite::Connection, label: &str) -> AppResult<Option<i64>> {
    let mut stmt = conn.prepare(
        "SELECT photo_id FROM ai_classifications WHERE label = ?1 LIMIT 1"

```rust
use crate::db::photo_dao;
use crate::thumbnail::generator;

/// Get the thumbnail path for the first photo with a given label.
#[tauri::command]
pub async fn cmd_get_label_preview(label: String, state: State<'_, AppState>) -> AppResult<String> {
    let conn = get_conn(state.clone())?;
    let project = {
        let guard = state.project.lock().unwrap();
        guard.as_ref().ok_or_else(|| AppError::Project("No project open".into()))?.clone()
    };

    let photo_id = classification_dao::get_label_preview_photo_id(&conn, &label)?
        .ok_or_else(|| AppError::NotFound(format!("No photos with label: {}", label)))?;

    let photo = photo_dao::get_photo_by_id(&conn, photo_id)?
        .ok_or_else(|| AppError::NotFound(format!("Photo {} not found", photo_id)))?;

    let out_dir = project.small_thumbnails_path();
    let thumb_path = generator::thumbnail_path_for(&photo.path, &out_dir);

    if thumb_path.exists() {
        Ok(thumb_path.to_string_lossy().to_string())
    } else {
        // Generate on demand
        generator::generate_small_thumbnail(Path::new(&photo.path), &out_dir)?;
        Ok(thumb_path.to_string_lossy().to_string())
    }
}
```

fn is_hidden(entry: &walkdir::DirEntry) -> bool {
    entry.file_name()
        .to_str()
        .map(|s| s.starts_with('.') || s == ".pf")
        .unwrap_or(false)
}
